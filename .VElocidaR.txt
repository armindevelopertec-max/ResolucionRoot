#if !defined(CONFIG_BT_ENABLED) || !defined(CONFIG_BLUEDROID_ENABLED)

#endif
BluetoothSerial SerialBT;
///////////////////// COMUNICACION SERIAL ////////////////
double posicion =0.0;
const int R=1980;
String inputString = "";
bool stringComplete = false;
const char separator = ',';
const int dataLength = 2;
int datos[dataLength];

//////////////////////MOTOR DERECHO/////////////////////////////// 
const int    C1L = 23;    // Entrada de la señal A del encoder C1(cable amarillo)
const int    C2L = 22;    // Entrada de la señal B del encoder C2(cable verde)

// Encoder
volatile int nL = 0;
volatile int antL      = 0;
volatile int actL      = 0;
volatile int N      = 0;
/////////////////////// TIEMPO DE MUESTREO /////////////////////
unsigned long lastTime = 0, sampleTime = 100;
//////// VARIABLES PARA CALCULAR VELOCIDADES ANGULARES /////////

// Configuracion de las salidas pwm
const int freq = 10000;
const int resolution = 8;

/////////////////////// INTERRUPCIONES //////////////////////
void IRAM_ATTR encoderL()
{
  antL=actL;
               
  if(digitalRead(C2L)) bitSet(actL,0); else bitClear(actL,0);
  if(digitalRead(C1L)) bitSet(actL,1); else bitClear(actL,1);
  
  
  if(antL == 2 && actL ==0) nL++;
  if(antL == 0 && actL ==1) nL++;
  if(antL == 3 && actL ==2) nL++;
  if(antL == 1 && actL ==3) nL++;
  
  if(antL == 1 && actL ==0) nL--;
  if(antL == 3 && actL ==1) nL--;
  if(antL == 0 && actL ==2) nL--;
  if(antL == 2 && actL ==3) nL--;    

}


void setup()
{
  SerialBT.begin("nukkuapa");  // Inicializa Bluetooth con el nombre del dispositivo
  SerialBT.println("Esperando comandos..."); 

  // Configuracion de los pines
  pinMode(C1L, INPUT);
  pinMode(C2L, INPUT);

  attachInterrupt(C1L, encoderL, CHANGE);
  attachInterrupt(C2L, encoderL, CHANGE);  
}

void V_Rpm(void)
{
  N = (nL*60.0*1000.0)/((millis()-lastTime)*R); // Calculamos las revoluciones por minuto
  lastTime = millis(); // Almacenar el tiempo actual.
  nL = 0;  // Resetear los pulsos.
} 
void loop() 
{
  if(SerialBT.available()) serialEvent();
  if(millis()-lastTime >= sampleTime)
  {
    V_Rpm();
    SerialBT.print("Velocidad en RPM Motor izquierda :");SerialBT.println(N);
  
  }
  
}

/////////////// RECEPCION DE DATOS /////////////////////
void serialEvent() {
  while (SerialBT.available()) {
    char inChar = (char)SerialBT.read();
    inputString += inChar;
    if (inChar == '\n') {
      stringComplete = true;
    }
  }
}
